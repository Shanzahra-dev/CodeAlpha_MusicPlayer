<script>
const $ = id => document.getElementById(id);
const audio = new Audio();
audio.preload = "metadata";
audio.volume = 0.8;

let songs = [];          // { name, url }
let index = -1;
let shuffle = false;

const fileInput = $("fileInput"), list = $("list"), seek = $("seek");
const titleEl = $("title"), statusEl = $("status"), playBtn = $("play"), disc = $("disc");

function fmt(s) {
  if (!isFinite(s)) return "0:00";
  return Math.floor(s / 60) + ":" + String(Math.floor(s % 60)).padStart(2, "0");
}

/* ---------- Themes (colours + light/dark) ---------- */
const themes = {
  rainbow: ["#ff65b7", "#8662ff", "#17112f"],
  ocean:   ["#32d4eb", "#237de5", "#092d49"],
  sunset:  ["#ff9c53", "#ed4b8c", "#412039"],
  forest:  ["#79e0a0", "#299c84", "#113b37"],
  midnight:["#5aa9ff", "#2b6cf0", "#0a1633"],
  galaxy:  ["#c084fc", "#6366f1", "#120a2e"],
  candy:   ["#ff8fcf", "#ffb86b", "#3a1631"],
  lava:    ["#ff6a3d", "#e11d48", "#2d0f10"],
  mint:    ["#6ee7b7", "#38bdf8", "#0c2f2e"],
  gold:    ["#ffc93c", "#f59e0b", "#2b2108"],
  neon:    ["#39ff88", "#00d4ff", "#071a14"]
};
let mode = "dark";

function applyTheme(name) {
  const t = themes[name] || themes.rainbow;
  const root = document.documentElement;
  root.style.setProperty("--a", t[0]);
  root.style.setProperty("--b", t[1]);
  if (mode === "dark") root.style.setProperty("--bg", t[2]); else root.style.removeProperty("--bg");
  try { localStorage.setItem("rb-theme", name); } catch (e) {}
}
function applyMode() {
  document.documentElement.setAttribute("data-mode", mode);
  $("modeBtn").textContent = mode === "dark" ? "☀" : "☾";
  applyTheme($("theme").value);
  try { localStorage.setItem("rb-mode", mode); } catch (e) {}
}
$("theme").addEventListener("change", () => applyTheme($("theme").value));
$("modeBtn").addEventListener("click", () => { mode = mode === "dark" ? "light" : "dark"; applyMode(); });

try {
  mode = localStorage.getItem("rb-mode") === "light" ? "light" : "dark";
  const saved = localStorage.getItem("rb-theme");
  if (saved && themes[saved]) $("theme").value = saved;
} catch (e) {}
applyMode();

/* ---------- Add music from device ---------- */
fileInput.addEventListener("change", () => {
  const files = Array.from(fileInput.files);
  if (!files.length) return;
  const firstNew = songs.length;
  files.forEach(f => songs.push({ name: f.name.replace(/\.[^.]+$/, ""), url: URL.createObjectURL(f) }));
  renderList();
  playSong(firstNew);
  fileInput.value = "";
});

/* ---------- Playback ---------- */
function playSong(i) {
  if (i < 0 || i >= songs.length) return;
  index = i;
  audio.src = songs[i].url;
  titleEl.textContent = songs[i].name;
  statusEl.textContent = "";
  renderList();
  audio.play().catch(err => { statusEl.textContent = "Tap ▶ to play (" + err.name + ")"; });
}
function togglePlay() {
  if (index === -1) {
    if (songs.length) playSong(0); else statusEl.textContent = "Add a song first.";
    return;
  }
  if (audio.paused) audio.play().catch(err => { statusEl.textContent = "Cannot play: " + err.name; });
  else audio.pause();
}
function nextSong() {
  if (!songs.length) return;
  let n = (index + 1) % songs.length;
  if (shuffle && songs.length > 1) { do { n = Math.floor(Math.random() * songs.length); } while (n === index); }
  playSong(n);
}
function prevSong() {
  if (!songs.length) return;
  if (audio.currentTime > 3) { audio.currentTime = 0; return; }
  playSong((index - 1 + songs.length) % songs.length);
}
playBtn.addEventListener("click", togglePlay);
$("next").addEventListener("click", nextSong);
$("prev").addEventListener("click", prevSong);

$("shuffle").addEventListener("click", function () {
  shuffle = !shuffle;
  this.classList.toggle("on", shuffle);
  this.setAttribute("aria-pressed", shuffle);
});
$("repeat").addEventListener("click", function () {
  audio.loop = !audio.loop;
  this.classList.toggle("on", audio.loop);
  this.setAttribute("aria-pressed", audio.loop);
});

/* ---------- Audio events ---------- */
audio.addEventListener("play",  () => { playBtn.textContent = "❚❚"; disc.classList.add("spin"); });
audio.addEventListener("pause", () => { playBtn.textContent = "▶";  disc.classList.remove("spin"); });
audio.addEventListener("ended", nextSong);
audio.addEventListener("loadedmetadata", () => { $("duration").textContent = fmt(audio.duration); });
audio.addEventListener("timeupdate", () => {
  $("current").textContent = fmt(audio.currentTime);
  if (audio.duration) seek.value = (audio.currentTime / audio.duration) * 100;
});
audio.addEventListener("error", () => {
  if (audio.getAttribute("src")) statusEl.textContent = "This file can't be played. Try MP3, M4A, WAV or OGG.";
});
seek.addEventListener("input", () => {
  if (audio.duration) audio.currentTime = (seek.value / 100) * audio.duration;
});
$("volume").addEventListener("input", e => { audio.volume = Number(e.target.value); });

/* ---------- Playlist ---------- */
function renderList() {
  list.innerHTML = "";
  if (!songs.length) {
    const li = document.createElement("li");
    li.className = "empty";
    li.textContent = "Add music from your device to start.";
    list.appendChild(li);
    return;
  }
  songs.forEach((s, i) => {
    const li = document.createElement("li");
    if (i === index) li.className = "playing";
    const art = document.createElement("div");
    art.className = "art";
    art.textContent = i === index ? "▶" : "♫";
    const name = document.createElement("div");
    name.className = "name";
    name.textContent = s.name;
    li.append(art, name);
    li.addEventListener("click", () => playSong(i));
    list.appendChild(li);
  });
}
